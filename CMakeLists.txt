cmake_minimum_required(VERSION 3.21)
project(RayTracer VERSION 2.0.0 LANGUAGES CXX)

option(RAYTRACER_ENABLE_SDL "Build the SDL desktop viewer" ON)
option(RAYTRACER_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
option(RAYTRACER_THREAD_SANITIZER "Enable ThreadSanitizer (separate build)" OFF)
if(RAYTRACER_SANITIZERS AND RAYTRACER_THREAD_SANITIZER)
    message(FATAL_ERROR "AddressSanitizer and ThreadSanitizer require separate builds.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(THREADS_PREFER_PTHREAD_FLAG TRUE)
find_package(Threads REQUIRED)

add_library(raytracer_core
    RayTracer/integrator.cpp RayTracer/scene.cpp
    RayTracer/renderer.cpp RayTracer/framebuffer.cpp)
target_include_directories(raytracer_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/RayTracer)
target_link_libraries(raytracer_core PUBLIC Threads::Threads)
if(MSVC)
    target_compile_options(raytracer_core PUBLIC /W4 /permissive-)
else()
    target_compile_options(raytracer_core PUBLIC -Wall -Wextra -Wpedantic)
endif()
if(RAYTRACER_SANITIZERS AND NOT MSVC)
    target_compile_options(raytracer_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(raytracer_core PUBLIC -fsanitize=address,undefined)
endif()
if(RAYTRACER_THREAD_SANITIZER AND NOT MSVC)
    target_compile_options(raytracer_core PUBLIC -fsanitize=thread -fno-omit-frame-pointer)
    target_link_options(raytracer_core PUBLIC -fsanitize=thread)
endif()

add_executable(raytracer RayTracer/main.cpp)
if(RAYTRACER_ENABLE_SDL)
    if(APPLE)
        find_package(SDL2 CONFIG QUIET PATHS "${CMAKE_CURRENT_SOURCE_DIR}/SDL2.framework/Resources/CMake")
    else()
        find_package(SDL2 CONFIG QUIET)
    endif()
    if(NOT TARGET SDL2::SDL2)
        message(FATAL_ERROR "SDL2 was not found. Install SDL2 or configure with -DRAYTRACER_ENABLE_SDL=OFF.")
    endif()
    add_library(raytracer_viewer RayTracer/sdl_window.cpp)
    target_link_libraries(raytracer_viewer PUBLIC raytracer_core SDL2::SDL2)
    target_link_libraries(raytracer PRIVATE raytracer_viewer)
    target_compile_definitions(raytracer PRIVATE RT_HAS_SDL=1)
    if(APPLE)
        set_property(TARGET raytracer APPEND PROPERTY BUILD_RPATH "${CMAKE_CURRENT_SOURCE_DIR}")
    endif()
else()
    target_link_libraries(raytracer PRIVATE raytracer_core)
    target_compile_definitions(raytracer PRIVATE RT_HAS_SDL=0)
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(raytracer_tests tests/test_main.cpp)
    target_link_libraries(raytracer_tests PRIVATE raytracer_core)
    target_compile_definitions(raytracer_tests PRIVATE RT_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
    add_test(NAME core COMMAND raytracer_tests)
    set_tests_properties(core PROPERTIES TIMEOUT 90)
    find_package(Python3 COMPONENTS Interpreter QUIET)
    if(Python3_FOUND)
        add_test(NAME cli_png COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_cli.py $<TARGET_FILE:raytracer>)
        set_tests_properties(cli_png PROPERTIES TIMEOUT 60)
    endif()
    if(RAYTRACER_ENABLE_SDL)
        add_executable(raytracer_viewer_tests tests/test_viewer.cpp)
        target_link_libraries(raytracer_viewer_tests PRIVATE raytracer_viewer)
        add_test(NAME viewer COMMAND raytracer_viewer_tests)
        set_tests_properties(viewer PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy" TIMEOUT 30)
        if(APPLE)
            set_property(TARGET raytracer_viewer_tests APPEND PROPERTY BUILD_RPATH "${CMAKE_CURRENT_SOURCE_DIR}")
        endif()
    endif()
endif()
